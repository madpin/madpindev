{{- /* Custom table render hook. - Wraps every table in a scrollable, keyboard-focusable container. - Flags long tables (>= 12 body rows) so the header can be frozen with CSS. - Detects image-gallery tables (cells containing figures/images) and marks them so they render as a clean grid instead of a data table. - Expresses column alignment as CSS classes, never inline styles, to satisfy the site's Content-Security-Policy and html-validate no-inline-style rule. */ -}} {{- $gallery := false -}} {{- if gt (len .TBody) 0 -}} {{- range (index .TBody 0) -}} {{- if findRE "<(img|figure|picture)" (.Text | string) 1 -}} {{- $gallery = true -}} {{- end -}} {{- end -}} {{- end -}} {{- $long := and (not $gallery) (ge (len .TBody) 12) -}} {{- $class := "table-scroll" -}} {{- if $long }}{{ $class = printf "%s is-long" $class }}{{ end -}} {{- if $gallery }}{{ $class = printf "%s is-gallery" $class }}{{ end -}}
